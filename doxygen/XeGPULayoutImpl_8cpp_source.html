83// Attach layout attributes to all vector-type operands of operations within
84// the given operation's region. Reports an error if any vector operand lacks
85// a layout attribute.
86bool xegpu::recoverTemporaryLayouts(Operation *rootOp) {
87 auto result = rootOp->walk([&](Operation *op) {
88 for (OpOperand &operand : op->getOpOperands()) {
89 // Layouts are needed for vector type only.
90 if (!isa<VectorType>(operand.get().getType()))
91 continue;
92 // Skip block arguments since they don't have defining ops to attach
93 // layout attributes to.
94 if (isa<BlockArgument>(operand.get()))
95 continue;
96 auto layout = xegpu::getDistributeLayoutAttr(operand.get());
97 if (!layout) {
98 op->emitWarning("Could not find layout attribute for operand ")
99 << operand.getOperandNumber() << " of operation " << op->getName();
100 continue;
101 }
102 xegpu::setTemporaryLayout(operand, layout);
103 }
104 return WalkResult::advance();
105 });
106 return !result.wasInterrupted();
107}

138/// Infers the source layout attribute for a broadcast operation given the
139/// result layout attribute, result shape, source shape.
140xegpu::DistributeLayoutAttr
141xegpu::inferBroadcastSourceLayout(xegpu::DistributeLayoutAttr resLayout,
142 ArrayRef<int64_t> resShape,
143 ArrayRef<int64_t> srcShape) {
144
145 SmallVector<int64_t> bcastDims;
146 auto returnLayout = resLayout;
147
148 // Handling broadcast from low-rank to high-rank (e.g., 1D to 2D) case.
149 int dimDiff = resShape.size() - srcShape.size();
150
151 if (dimDiff > 0) {
152 // Adding the missing leading dims
153 for (int i = 0; i < dimDiff; i++)
154 bcastDims.push_back(i);
155
156 // Create a slice layout for the source
157 returnLayout = xegpu::SliceAttr::get(
158 resLayout.getContext(), resLayout,
159 DenseI64ArrayAttr::get(resLayout.getContext(), bcastDims));
160 }
161 return returnLayout;
162}

189/// Infers the source layout attribute for a bitcast operation given the
190/// result layout attribute, result element type bitwidth, and source element
191/// type bitwidth.
192xegpu::DistributeLayoutAttr
193xegpu::inferBitCastSourceLayout(xegpu::DistributeLayoutAttr resLayout,
194 int resElemTyBitWidth, int srcElemTyBitWidth) {
195
196 SmallVector<int64_t> sgData = resLayout.getEffectiveSgDataAsInt();
197 SmallVector<int64_t> instData = resLayout.getEffectiveInstDataAsInt();
198 SmallVector<int64_t> laneData = resLayout.getEffectiveLaneDataAsInt();
199 size_t sgDataSize = sgData.size();
200 size_t instDataSize = instData.size();
201 size_t laneDataSize = laneData.size();
202 int64_t sgDataValue = -1;
203 int64_t instDataValue = -1;
204 int64_t laneDataValue = -1;
205 int64_t dim = resLayout.getRank() - 1;
206
207 if (srcElemTyBitWidth <= resElemTyBitWidth) {
208 int bitWidthRatio = resElemTyBitWidth / srcElemTyBitWidth;
209 if (sgDataSize)
210 sgDataValue = sgData.back() * bitWidthRatio;
211 if (instDataSize)
212 instDataValue = instData.back() * bitWidthRatio;
213 if (laneDataSize)
214 laneDataValue = laneData.back() * bitWidthRatio;
215 } else {
216 int bitWidthRatio = srcElemTyBitWidth / resElemTyBitWidth;
217 if (sgDataSize) {
218 assert((sgData.back() % bitWidthRatio) == 0 &&
219 "sgData not divisible by bitWidthRatio");
220 sgDataValue = sgData.back() / bitWidthRatio;
221 }
222 if (instDataSize) {
223 assert((instData.back() % bitWidthRatio) == 0 &&
224 "instData not divisible by bitWidthRatio");
225 instDataValue = instData.back() / bitWidthRatio;
226 }
227 if (laneDataSize) {
228 assert((laneData.back() % bitWidthRatio) == 0 &&
229 "laneData not divisible by bitWidthRatio");
230 laneDataValue = laneData.back() / bitWidthRatio;
231 }
232 }
233
234 xegpu::DistributeLayoutAttr finalSrcLayout;
235 finalSrcLayout =
236 resLayout.setDimData(dim, sgDataValue, instDataValue, laneDataValue);
237
238 return finalSrcLayout;
239}

241/// Infers the source layout attribute for an insert strided slice operation
242/// given the result layout attribute, result shape, and source shape. Removes
243/// leading dimensions from the result layout to match the source shape size.
244xegpu::DistributeLayoutAttr xegpu::inferInsertStridedSliceSourceLayout(
245 xegpu::DistributeLayoutAttr resLayout, ArrayRef<int64_t> resShape,
246 ArrayRef<int64_t> srcShape) {
247
248 int srcShapeSize = srcShape.size();
249 int resShapeSize = resShape.size();
250 int dimDiff = resShapeSize - srcShapeSize;
251
252 if (dimDiff > 0) {
253 // assert that the leading dimensions being sliced off are not distributed
254 // (i.e. sg_layout and lane_layout for those dimensions are all 1)
255 auto resSgLayout = resLayout.getEffectiveSgLayoutAsInt();
256 auto resLaneLayout = resLayout.getEffectiveLaneLayoutAsInt();
257 for (int i = 0; i < dimDiff; i++) {
258 assert((resSgLayout.size() == 0 || resSgLayout[i] == 1) &&
259 (resLaneLayout.size() == 0 || resLaneLayout[i] == 1) &&
260 "Leading dimensions being sliced off must not be distributed");
261 }
262 return resLayout.dropDims(llvm::to_vector(llvm::seq<int64_t>(0, dimDiff)));
263 }
264 return resLayout;
265}

267/// Infers the source layout attribute for a shape cast operation given the
268/// result layout attribute, result shape, and source shape.
269xegpu::DistributeLayoutAttr
270xegpu::inferShapeCastSourceLayout(xegpu::DistributeLayoutAttr resLayout,
271 ArrayRef<int64_t> resShape,
272 ArrayRef<int64_t> srcShape) {
273
274 // There are three use cases:
275 // 1. expand dims of low-rank dimensions (e.g., 1D to 2D): to set up the
276 // tensor before broadcast
277 // 2. split dim of a high-rank dimension (e.g., 1D to 2D): to setup tensor
278 // for multi-stage reduction
279 // 3. combines all dims to a single dim and put in the innermost dim in 2d as
280 // [1, combinedData] or [combinedData]. Say, [2, 4, 8] -> [1, 64] or [64]
281 // Use cases are only supported after workgroup distribution,
282 // like cross-sg reduction saves multidimension data to
283 // 1D slm buffer, shapecast inserted by cse/canonicalization passes.
284
285 // Use case 1: Shapes only differ by expanding unit dimensions, for broadcast
286 SmallVector<int64_t> expandedUnitDims;
287
288 if (xegpu::matchUnitDimExpansion(srcShape, resShape, expandedUnitDims)) {
289 // create a slice layout for the source by removing the expanded unit dims
290 auto sliceDimsAttr = DenseI64ArrayAttr::get(
291 resLayout.getContext(), ArrayRef<int64_t>(expandedUnitDims));
292 auto srcLayout =
293 xegpu::SliceAttr::get(resLayout.getContext(), resLayout, sliceDimsAttr);
294 return srcLayout;
295 }
296
297 // Use case 2: Dim split from source to result, for multi-stage reduction
298 SmallVector<SmallVector<int64_t>> splitDimGroups;
299 if (xegpu::matchSplitDimExpansion(srcShape, resShape, splitDimGroups)) {
300 auto srcLayout = resLayout;
301 for (const auto &dimGroup : splitDimGroups)
302 srcLayout = srcLayout.collapseDims(dimGroup);
303
304 return srcLayout;
305 }
306
307 // Use case 3: Collaspse to innermost dim, for cross-sg reduction to SLM
308 auto matchCollapseToInnermostDim = [&](ArrayRef<int64_t> src,
309 ArrayRef<int64_t> dst) -> bool {
310 // only one non-unit dim in dst which is the innermost dim
311 if ((dst.size() != 2) && (dst.size() != 1))
312 return false;
313 int64_t srcSize = std::accumulate(src.begin(), src.end(), 1LL,
314 std::multiplies<int64_t>());
315 if (dst.size() == 1)
316 return (dst[0] == srcSize);
317 return (dst[0] == 1) && (dst[1] == srcSize);
318 };
319
320 if (matchCollapseToInnermostDim(srcShape, resShape)) {
321 int srcShapeSize = srcShape.size();
322 int resShapeSize = resShape.size();
323 auto context = resLayout.getContext();
324 auto resInstData = resLayout.getEffectiveInstDataAsInt();
325 auto resLaneLayout = resLayout.getEffectiveLaneLayoutAsInt();
326 auto resLaneData = resLayout.getEffectiveLaneDataAsInt();
327
328 // Extract layout info from result's innermost dimension and apply to
329 // source's innermost dimension while setting all other dimensions to 1.
330 // The inferred layout is restricted by srcShape to ensure it fits within
331 // the source dimensions.
332 // Examples 1:
333 // srcShape=[8, 16, 32], resShape=[1, 4096]
334 // resInstData=[1, 16]
335 // -> inferredInstData=[1, 1, min(16, 32)]=[1, 1, 16]
336 // Examples 2:
337 // srcShape=[4, 8, 64], resShape=[2048]
338 // resLaneLayout=[16], resLaneData=[2]
339 // -> inferredLaneLayout=[1, 1, 16]
340 // -> inferredLaneData=[1, 1, min(2, 64/16)]=[1, 1, 2]
341
342 if (resInstData.size() != 0) {
343 // assert resInstData must be 1 for all but the innermost dim
344 for (int i = 0; i < resShapeSize - 1; i++) {
345 assert(resInstData[i] == 1 &&
346 "only innermost dim can have non-unit instData");
347 }
348 SmallVector<int> inferredInstData(srcShapeSize, 1);
349 inferredInstData[srcShapeSize - 1] =
350 std::min(resInstData[resShapeSize - 1], srcShape[srcShapeSize - 1]);
351 return xegpu::LayoutAttr::get(context, inferredInstData);
352 }
353
354 if (resLaneLayout.size() != 0) {
355 for (int i = 0; i < resShapeSize - 1; i++) {
356 assert(resLaneData[i] == 1 &&
357 "only innermost dim can have non-unit instData");
358 }
359 assert(srcShape.back() % resLaneLayout.back() == 0 &&
360 "source innermost dim must be >= result lane layout");
361 SmallVector<int> inferredLaneLayout(srcShapeSize, 1);
362 SmallVector<int> inferredLaneData(srcShapeSize, 1);
363 inferredLaneLayout.back() = resLaneLayout.back();
364 inferredLaneData.back() = std::min(
365 resLaneData.back(), srcShape.back() / inferredLaneLayout.back());
366 return xegpu::LayoutAttr::get(context, inferredLaneLayout,
367 inferredLaneData);
368 }
369 }
370 llvm_unreachable("running into unsupported shape cast scenarios");
371 return nullptr;
372}

374/// Sets up layout for reduction operations by creating a SliceAttr for the
375/// result.
376///
377/// Algorithm Overview:
378/// This function attempts to construct a source layout that, when sliced along
379/// reduction dimensions, produces a result layout compatible with the
380/// consumer layout.
381///
382/// For subgroup layouts, it first tries to align the source layout's subgroup
383/// layout and data with the consumer's layout on non-reduction dimensions.
384/// Then, it distributes remaining subgroups across reduction dimensions. This
385/// avoids subgroup data redistribution overhead between the reduced result and
386/// its consumer.
387///
388/// InstData requries {1, ..., min(maxReduceVectorSize, srcShape),subgroupSize}
389/// Lane Layout requires {1, ..., 1, subgroupSize}
390/// Lane data requires {1, ..., min(maxReduceVectorSize, srcShape), 1}
391///
392/// Examples:
393/// 1. Subgroup layout - Row reduction on 2D tensor:
394/// srcShape=[32, 64], reductionDims=[1], resShape=[32], subgroupSize=16,
395/// workgroupSize=32
396/// Consumer Layout:
397/// #xegpu.slice<#xegpu.layout<sg_layout=[4, 8], sg_data=[8, 8]>, dims =
398/// [1]>} Result: srcLayout with sgLayout=[4, 8], sgData=[8, 8] (matches
399/// consumer on non-reduction dim, minimizing data redistribution on
400/// reduction dim)
401/// 2. Subgroup layout - Same example above but consumer has different layout:
402/// sgLayout=[32], sgData=[1]
403/// Result: srcLayout with sgLayout=[32,1], sgData=[1, 64]
404/// (distributes all subgroups on non reduction dim)
405///
406/// 2. InstData layout - Column reduction:
407/// srcShape=[32, 64], reductionDims=[0], subgroupSize=16
408/// Result: instData=[1, 16] (maxReduceVectorSize=1, subgroupSize on
409/// innermost)
410///
411/// 3. Lane layout - Multi-dimensional reduction:
412/// srcShape=[16, 32, 64], reductionDims=[1], subgroupSize=16
413/// Result: laneLayout=[1, 1, 16], laneData=[1, 1, 1]
414/// (subgroupSize on innermost dim, max vector size on reduction dim)
415
416xegpu::SliceAttr xegpu::setupMultiReductionResultLayout(
417 xegpu::LayoutKind layoutKind, VectorType srcVecTy,
418 DistributeLayoutAttr consumerLayout, SmallVector<int64_t> reductionDims,
419 const xegpu::uArch::uArch *uArch) {
420
421 auto srcShape = srcVecTy.getShape();
422 int srcRank = srcShape.size();
423 auto context = consumerLayout.getContext();
424
425 // Reduction layout requires at least 2D tensors
426 if (srcRank < 2)
427 return nullptr;
428
429 // Helper lambda to convert int64 vectors to int32 DenseArrayAttr
430 auto toInt32Attr = [&](ArrayRef<int64_t> vec) {
431 SmallVector<int32_t> vec32(vec.begin(), vec.end());
432 return DenseI32ArrayAttr::get(context, vec32);
433 };
434
435 // Extract original plain layout for workgroup/subgroup size recovery
436 xegpu::SliceAttr consumerSliceLayout =
437 dyn_cast<xegpu::SliceAttr>(consumerLayout);
438 DistributeLayoutAttr plainLayout =
439 consumerSliceLayout ? consumerSliceLayout.flatten().getParent()
440 : consumerLayout;
441
442 const int subgroupSize = uArch->getSubgroupSize();
443 int64_t maxReduceVectorSize = 1; // could extend to spirv vector Size
444
445 xegpu::DistributeLayoutAttr srcLayout;
446
447 if (layoutKind == xegpu::LayoutKind::Subgroup) {
448 auto sgLayoutVec = plainLayout.getEffectiveSgLayoutAsInt();
449 const int workgroupSize = std::accumulate(
450 sgLayoutVec.begin(), sgLayoutVec.end(), 1, std::multiplies<int64_t>());
451 SmallVector<int64_t> sgLayout(srcRank), sgData(srcRank);
452 SmallVector<int64_t> consumerSgLayout =
453 consumerLayout.getEffectiveSgLayoutAsInt();
454 int remainingSgCount = workgroupSize;
455 int consumerIdx = consumerSgLayout.size() - 1;
456
457 // First pass: Match consumer's layout on non-reduction dimensions
458 for (int i = srcRank - 1; i >= 0; i--) {
459 if (!llvm::is_contained(reductionDims, i) && consumerIdx >= 0) {
460 sgLayout[i] = consumerSgLayout[consumerIdx];
461 assert((srcShape[i] % sgLayout[i] == 0) &&
462 "source shape not divisible by consumer sg_layout");
463 sgData[i] = srcShape[i] / sgLayout[i];
464 remainingSgCount /= sgLayout[i];
465 consumerIdx--;
466 }
467 }
468
469 // Second pass: Distribute remaining subgroups across reduction dimensions
470 for (int i = srcRank - 1; i >= 0; i--) {
471 if (llvm::is_contained(reductionDims, i)) {
472 sgLayout[i] =
473 std::min(srcShape[i], static_cast<int64_t>(remainingSgCount));
474 assert((srcShape[i] % sgLayout[i] == 0) &&
475 "source shape not divisible by sg_layout");
476 sgData[i] = srcShape[i] / sgLayout[i];
477 remainingSgCount /= sgLayout[i];
478 }
479 }
480
481 assert(remainingSgCount == 1 && "not all subgroups distributed");
482 srcLayout = xegpu::LayoutAttr::get(
483 context, toInt32Attr(sgLayout), toInt32Attr(sgData),
484 /*inst_data =*/nullptr, /*lane_layout =*/nullptr,
485 /*lane_data =*/nullptr, /*order =*/nullptr);
486
487 } else if (layoutKind == xegpu::LayoutKind::InstData) {
488
489 SmallVector<int64_t> instData(srcRank, 1);
490 instData[srcRank - 2] =
491 std::min(maxReduceVectorSize, srcShape[srcRank - 2]);
492 instData[srcRank - 1] = subgroupSize;
493 srcLayout = xegpu::LayoutAttr::get(context, toInt32Attr(instData));
494
495 } else if (layoutKind == xegpu::LayoutKind::Lane) {
496
497 SmallVector<int64_t> laneLayout(srcRank, 1), laneData(srcRank, 1);
498 laneLayout[srcRank - 1] = subgroupSize;
499 laneData[srcRank - 2] =
500 std::min(maxReduceVectorSize, srcShape[srcRank - 2]);
501 srcLayout = xegpu::LayoutAttr::get(context, toInt32Attr(laneLayout),
502 toInt32Attr(laneData),
503 consumerLayout.getOrder());
504 }
505
506 return xegpu::SliceAttr::get(context, srcLayout,
507 DenseI64ArrayAttr::get(context, reductionDims));
508}

510/// Sets up the result layout for a bitcast operation.
511/// When casting to a smaller bitwidth, adjusts the layout dimensions (sgData,
512/// instData, or laneData) by multiplying by the bitwidth ratio to ensure the
513/// result layout can be correctly divided back to the source layout during
514/// inference.
515///
516/// Examples:
517/// 1. Casting f32 -> f16 (32-bit to 16-bit, bitWidthRatio = 2):
518/// Consumer layout: instData=[1, 16], subgroupSize=16
519/// Source shape: [8, 32]
520/// Result layout: instData=[1, 32] (16 * 2)
521/// The innermost dimension is multiplied by 2 to maintain consistency.
522///
523/// 2. Casting f32 -> i8 (32-bit to 8-bit, bitWidthRatio = 4):
524/// Consumer instData=[1, 16], subgroupSize=16
525/// Source shape: [4, 128]
526/// adjust the instData from [1, 16] to [1, 16 * 4 = 64]
527///
528/// 3. Casting i8 -> i32 (8-bit to 32-bit, bitWidthRatio = 1/4):
529/// Consumer layout: laneLayout=[1, 16], laneData=[1, 4]
530/// No adjustment needed - returns consumer layout directly.
531///
532xegpu::DistributeLayoutAttr xegpu::setupBitCastResultLayout(
533 xegpu::LayoutKind layoutKind, VectorType srcVecTy, VectorType resVecTy,
534 DistributeLayoutAttr consumerLayout, const xegpu::uArch::uArch *uArch) {
535
536 int srcElemTyBitWidth = srcVecTy.getElementType().getIntOrFloatBitWidth();
537 int resElemTyBitWidth = resVecTy.getElementType().getIntOrFloatBitWidth();
538
539 ArrayRef<int64_t> srcShape = srcVecTy.getShape();
540 SmallVector<int64_t> sgData = consumerLayout.getEffectiveSgDataAsInt();
541 SmallVector<int64_t> instData = consumerLayout.getEffectiveInstDataAsInt();
542 SmallVector<int64_t> laneData = consumerLayout.getEffectiveLaneDataAsInt();
543 assert(consumerLayout.getRank() == static_cast<int64_t>(srcShape.size()) &&
544 "laneData must be available for all dimensions");
545 size_t dim = srcShape.size() - 1;
546 int64_t sgDataValue = -1;
547 int64_t instDataValue = -1;
548 int64_t laneDataValue = -1;
549 const int subgroupSize = uArch->getSubgroupSize();
550
551 if (srcElemTyBitWidth > resElemTyBitWidth) {
552 // When casting to a smaller bitwidth, multiply the result layout
553 // accordingly to ensure it can be divided by the ratio back to the
554 // source layout.
555 int bitWidthRatio = srcElemTyBitWidth / resElemTyBitWidth;
556 int innermostDimLaneLayout = subgroupSize;
557 if (layoutKind == xegpu::LayoutKind::Subgroup) {
558 sgDataValue = sgData[dim];
559 } else if (layoutKind == xegpu::LayoutKind::InstData) {
560 instDataValue = instData[dim];
561 // Adjust instDataValue so it still fits within an instruction after
562 // dividing by bitWidthRatio
563 while ((instDataValue <= srcShape[dim]) &&
564 (instDataValue % (innermostDimLaneLayout * bitWidthRatio) != 0))
565 instDataValue *= 2;
566 assert((srcShape[dim] % instDataValue) == 0 &&
567 "srcShape, instData, and lanelayout for innermost must be 2^n !");
568 } else if (layoutKind == xegpu::LayoutKind::Lane) {
569 laneDataValue = laneData[dim];
570 while ((laneDataValue <= srcShape[dim]) &&
571 (laneDataValue % bitWidthRatio != 0))
572 laneDataValue *= 2;
573 }
574 // Now set only instData and laneData, preserving sgData
575 xegpu::DistributeLayoutAttr resLayout;
576 resLayout = consumerLayout.setDimData(dim, sgDataValue, instDataValue,
577 laneDataValue);
578 return resLayout;
579 }
580 return consumerLayout;
581}

583/// Sets up the result layout for an insert strided slice operation.
584/// Creates a result layout based on the specified layout kind (InstData or
585/// Lane).
586xegpu::DistributeLayoutAttr xegpu::setupInsertStridedSliceResultLayout(
587 xegpu::LayoutKind layoutKind, VectorType srcVectorTy,
588 VectorType resVectorTy, xegpu::DistributeLayoutAttr consumerLayout,
589 const xegpu::uArch::uArch *uArch) {
590
591 xegpu::DistributeLayoutAttr requiredResLayout;
592 SmallVector<int64_t> consumerInstData =
593 consumerLayout.getEffectiveInstDataAsInt();
594 SmallVector<int64_t> consumerLaneData =

849// This function returns all layouts for the given sgCount, whose sgData:
850// 1. Evenly divides the wgShape.
851// 2. Is a multiple of instData.
852// Example:
853// wgShape = [128, 64], instData = [8, 16], sgCount = 32
854// Returns layouts:
855// [(8,4), (16,2)], which correspond to sgData [16,16] and [8,32].
856using LayoutRepresentation = std::pair<int64_t, int64_t>;
857static SmallVector<LayoutRepresentation>
858getValidLayouts(ArrayRef<int64_t> wgShape, ArrayRef<int64_t> instData,
859 int64_t sgCount) {
860 SmallVector<LayoutRepresentation> candidates;
861 for (int sgLayout0 = 1; sgLayout0 <= sgCount; ++sgLayout0) {
862 if (sgCount % sgLayout0)
863 continue;
864 int64_t sgLayout1 = sgCount / sgLayout0;
865 int64_t sgData0 = wgShape[0] / sgLayout0;
866 int64_t sgData1 = wgShape[1] / sgLayout1;
867 if ((wgShape[0] % sgLayout0 || wgShape[1] % sgLayout1) ||
868 (sgData0 % instData[0] || sgData1 % instData[1]))
869 continue;
870 candidates.emplace_back(sgLayout0, sgLayout1);
871 }
872 // Sort primarily by how balanced they are
873 // (i.e., minimize the absolute difference between the two dimensions), and
874 // secondarily by the first dimension in ascending order.
875 llvm::sort(candidates, [](const LayoutRepresentation &lhs,
876 const LayoutRepresentation &rhs) {
877 int diffLhs = std::abs(lhs.first - lhs.second);
878 int diffRhs = std::abs(rhs.first - rhs.second);
879 if (diffLhs != diffRhs)
880 return diffLhs < diffRhs;
881 return lhs.first < rhs.first;
882 });
883 return candidates;
884}

886/// Sets up the anchor layouts for dpas operands (A, B, and C/D).
887/// The numSg and consumerLayout (optional) are only used by sg layout
888/// creation.
889std::optional<
890 std::tuple<xegpu::DistributeLayoutAttr, xegpu::DistributeLayoutAttr,
891 xegpu::DistributeLayoutAttr>>
892xegpu::setupDpasLayout(xegpu::LayoutKind layoutKind, VectorType aTy,
893 VectorType bTy, VectorType cdTy,
894 xegpu::DistributeLayoutAttr consumerLayout,
895 const xegpu::uArch::uArch *uArch, int numSg) {
896 auto context = aTy.getContext();
897 const auto *uArchInstruction =
898 dyn_cast<xegpu::uArch::SubgroupMatrixMultiplyAcc>(uArch->getInstruction(
899 xegpu::uArch::InstructionKind::SubgroupMatrixMultiplyAcc));
900
901 auto getInstDataVectors = [&]()
902 -> std::optional<std::tuple<SmallVector<int64_t>, SmallVector<int64_t>,
903 SmallVector<int64_t>>> {
904 const int subgroupSize = uArch->getSubgroupSize();
905 const unsigned dataALen = aTy.getShape().front();
906 auto supportedALen = uArchInstruction->getSupportedM(aTy.getElementType());
907 const int maxALen =
908 xegpu::getLargestDivisor(dataALen, ArrayRef<unsigned>(supportedALen));
909
910 const unsigned dataBLen = bTy.getShape().back();
911 auto supportedBLen = uArchInstruction->getSupportedN(bTy.getElementType());
912 const int maxBLen =
913 xegpu::getLargestDivisor(dataBLen, ArrayRef<unsigned>(supportedBLen));
914
915 auto supportedCLen = uArchInstruction->getSupportedN(cdTy.getElementType());
916 const int maxCLen =
917 xegpu::getLargestDivisor(dataBLen, ArrayRef<unsigned>(supportedCLen));
918 if (maxALen == -1 || maxBLen == -1 || maxCLen == -1)
919 return std::nullopt;
920
921 SmallVector<int64_t> instDataA(aTy.getRank(), 1);
922 instDataA[aTy.getRank() - 2] = maxALen;
923 instDataA[aTy.getRank() - 1] = subgroupSize;
924 SmallVector<int64_t> instDataB(bTy.getRank(), 1);
925 instDataB[bTy.getRank() - 2] = subgroupSize;
926 instDataB[bTy.getRank() - 1] = maxBLen;
927 SmallVector<int64_t> instDataCD(cdTy.getRank(), 1);
928 instDataCD[cdTy.getRank() - 2] = maxALen;
929 instDataCD[cdTy.getRank() - 1] = maxCLen;
930 return std::make_tuple(instDataA, instDataB, instDataCD);
931 };
932
933 if (layoutKind == xegpu::LayoutKind::Subgroup) {
934 assert(numSg > 0 &&
935 "Number of subgroups must be provided for sg layout creation.");
936 auto instDataVecs = getInstDataVectors();
937 if (!instDataVecs)
938 return std::nullopt;
939 auto [instDataA, instDataB, instDataCD] = *instDataVecs;
940 assert(instDataA.size() == 2 && instDataB.size() == 2 &&
941 instDataCD.size() == 2 &&
942 "Sg layout creation expects valid 2D inst data");
943
944 std::optional<LayoutRepresentation> consumerSgLayout = std::nullopt;
945 if (consumerLayout && consumerLayout.isForWorkgroup()) {
946 SmallVector<int64_t> sgLayoutD =
947 consumerLayout.getEffectiveSgLayoutAsInt();
948 consumerSgLayout = std::make_pair(sgLayoutD[0], sgLayoutD[1]);
949 }
950
951 // Step 1. Get all valid layouts for A, B and C/D operands.
952 // Order them from most balanced to least balanced.
953 auto layoutsA = getValidLayouts(aTy.getShape(), instDataA, numSg);
954 auto layoutsB = getValidLayouts(bTy.getShape(), instDataB, numSg);
955 auto layoutsCD = getValidLayouts(cdTy.getShape(), instDataCD, numSg);
956 if (layoutsA.empty() || layoutsB.empty() || layoutsCD.empty())
957 return std::nullopt;
958
959 // Step 2. If the consumer layout can be reused for all operands, that
960 // layout is chosen. Otherwise, pick the most balanced subgroup layout
961 // that is valid for A, B and C (if present) operands
962 llvm::DenseSet<LayoutRepresentation> setA(layoutsA.begin(), layoutsA.end());
963 llvm::DenseSet<LayoutRepresentation> setCD(layoutsCD.begin(),
964 layoutsCD.end());
965 std::optional<LayoutRepresentation> bestPick;
966 for (auto &sgLayout : layoutsB) {
967 if (setA.contains(sgLayout) && setCD.contains(sgLayout)) {
968 // Is in (A and B and CD) and matches consumer -> best pick
969 if (consumerSgLayout.has_value() && sgLayout == *consumerSgLayout) {
970 bestPick = sgLayout;
971 break;
972 }
973 // Is in (A and B and CD) layoutsB is ordered from most
974 // balanced to least. So the first one we see is the most balanced
975 // one, remember it and later only update if there is one that matches
976 // the consumer.
977 if (!bestPick)
978 bestPick = sgLayout;
979 }
980 }
981 // Step 3. If there is no subgroup layout compatible with A, B and C (if
982 // present) operands, we fail.
983 if (!bestPick)
984 return std::nullopt;
985 SmallVector<int> sgLayout = {static_cast<int>(bestPick->first),
986 static_cast<int>(bestPick->second)};
987 SmallVector<int> sgDataA = {
988 static_cast<int>(aTy.getShape()[0] / sgLayout[0]),
989 static_cast<int>(aTy.getShape()[1] / sgLayout[1])};
990 SmallVector<int> sgDataB = {
991 static_cast<int>(bTy.getShape()[0] / sgLayout[0]),
992 static_cast<int>(bTy.getShape()[1] / sgLayout[1])};
993 SmallVector<int> sgDataCD = {
994 static_cast<int>(cdTy.getShape()[0] / sgLayout[0]),
995 static_cast<int>(cdTy.getShape()[1] / sgLayout[1])};
996
997 auto dpasALayout = xegpu::LayoutAttr::get(
998 context, DenseI32ArrayAttr::get(context, sgLayout),
999 DenseI32ArrayAttr::get(context, sgDataA),
1000 /*inst_data =*/nullptr, /*lane_layout =*/nullptr,
1001 /*lane_data =*/nullptr, /*order =*/nullptr);
1002
1003 auto dpasBLayout = xegpu::LayoutAttr::get(
1004 context, DenseI32ArrayAttr::get(context, sgLayout),
1005 DenseI32ArrayAttr::get(context, sgDataB),
1006 /*inst_data =*/nullptr, /*lane_layout =*/nullptr,
1007 /*lane_data =*/nullptr, /*order =*/nullptr);
1008
1009 auto dpasCDLayout = xegpu::LayoutAttr::get(
1010 context, DenseI32ArrayAttr::get(context, sgLayout),
1011 DenseI32ArrayAttr::get(context, sgDataCD),
1012 /*inst_data =*/nullptr, /*lane_layout =*/nullptr,
1013 /*lane_data =*/nullptr, /*order =*/nullptr);
1014 return std::make_tuple(dpasALayout, dpasBLayout, dpasCDLayout);
1015 } else if (layoutKind == xegpu::LayoutKind::InstData) {
1016 auto instDataVecs = getInstDataVectors();
1017 if (!instDataVecs)
1018 return std::nullopt;
1019 auto [instDataA, instDataB, instDataCD] = *instDataVecs;
1020 return std::make_tuple(
1021 xegpu::LayoutAttr::get(
1022 context, SmallVector<int>(instDataA.begin(), instDataA.end())),
1023 xegpu::LayoutAttr::get(
1024 context, SmallVector<int>(instDataB.begin(), instDataB.end())),
1025 xegpu::LayoutAttr::get(
1026 context, SmallVector<int>(instDataCD.begin(), instDataCD.end())));
1027 } else if (layoutKind == xegpu::LayoutKind::Lane) {
1028 auto aLayout = getDefaultLaneLayout2DBlockIo(
1029 aTy, uArch, uArchInstruction->getPackedFormatBitSizeA());
1030 auto bLayout = getDefaultLaneLayout2DBlockIo(
1031 bTy, uArch, uArchInstruction->getPackedFormatBitSizeB(), true);
1032 auto cdLayout = getDefaultLaneLayout2DBlockIo(
1033 cdTy, uArch, uArchInstruction->getPackedFormatBitSizeB());
1034 return std::make_tuple(aLayout, bLayout, cdLayout);
1035 }
1036 return std::nullopt;
1037}

1048 if (!resLayout)

1051 if (!srcTy)

1059 // using reduction dims. Acc operand is expected to have the same layout as
1060 // the result.

1073 // element type bitwidths.

1088 if (!resLayout)

1102 resLayout, insertSlice.getDestVectorType().getShape(),
